InterruptPropagation.java
public class InterruptPropagation {
    static void subTask() throws InterruptedException { Thread.currentThread().interrupt(); Thread.sleep(1000); }
    public static void main(String[] args) {
        Runnable task = () -> {
            try { subTask(); }
            catch (InterruptedException e) { Thread.currentThread().interrupt(); System.out.println("cancelled=" + Thread.currentThread().isInterrupted()); }
            finally { System.out.println("released"); }
        };
        task.run(); Thread.interrupted();
    }
}
InterruptWaitRoute.java
import java.util.concurrent.CountDownLatch;
public class InterruptWaitRoute {
    public static void main(String[] args) throws InterruptedException {
        CountDownLatch ready = new CountDownLatch(1), neverOpen = new CountDownLatch(1);
        Thread worker = new Thread(() -> {
            try { ready.countDown(); neverOpen.await(); }
            catch (InterruptedException e) {
                System.out.println("catch=" + Thread.currentThread().isInterrupted()); Thread.currentThread().interrupt();
                System.out.println("restored=" + Thread.currentThread().isInterrupted());
            } finally { System.out.println("cleanup"); }
        }, "waiting-worker");
        worker.start(); ready.await(); worker.interrupt(); worker.join(); System.out.println("done");
    }
}
StartVersusRun.java
public class StartVersusRun {
    public static void main(String[] args) throws InterruptedException {
        Runnable task = () -> System.out.println(Thread.currentThread().getName().equals("main") ? "run:main" : "start:helper");
        task.run();
        Thread worker = new Thread(task, "helper"); worker.start(); worker.join(); System.out.println("joined");
    }
}
